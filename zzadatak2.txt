using System;
using System.Collections.Generic;
using System.ComponentModel;
using System.Data;
using System.Drawing;
using System.Linq;
using System.Text;
using System.Threading.Tasks;
using System.Windows.Forms;

namespace opensave
{
    public partial class Form1 : Form
    {
        bool crtam = false;
        Point prethodnatacka;
        Point pocetnatacka;
        // Boja linije
        Color bojaLinije = Color.Red;

        // Boja tačke / olovke
        Color bojaTacke = Color.Cyan;

        public Form1()
        {
            InitializeComponent();
        }

        private void button1_Click(object sender, EventArgs e)
        {
            openFileDialog1.InitialDirectory = @"D:\slike";
            openFileDialog1.Title = "Izaberi sliku...";
            openFileDialog1.DefaultExt = "jpg";
            openFileDialog1.Filter = "Slike (*.jpg)|*.jpg";
            openFileDialog1.RestoreDirectory = true;
            openFileDialog1.ShowDialog();
        }

        private void openFileDialog1_FileOk(object sender, CancelEventArgs e)
        {
            textBox1.Text = openFileDialog1.FileName;
            pictureBox1.BackgroundImage = Image.FromFile(openFileDialog1.FileName);
        }

        private void button2_Click(object sender, EventArgs e)
        {
            saveFileDialog1.InitialDirectory = @"D:\slike";
            saveFileDialog1.Title = "Izaberi sliku...";
            saveFileDialog1.DefaultExt = "jpg";
            saveFileDialog1.Filter = "Slike (*.jpg)|*.jpg";
            saveFileDialog1.RestoreDirectory = true;
            saveFileDialog1.ShowDialog();
        }

        private void saveFileDialog1_FileOk(object sender, CancelEventArgs e)
        {
            string s = saveFileDialog1.FileName;
            pictureBox1.BackgroundImage.Save(s);
        }

        private void button3_Click(object sender, EventArgs e)
        {
            using (Graphics g= Graphics.FromImage(pictureBox1.BackgroundImage))
            {
                int x1 = Convert.ToInt16(textBox2.Text);
                int y1 = Convert.ToInt16(textBox3.Text);
                int x2 = Convert.ToInt16(textBox4.Text);
                int y2 = Convert.ToInt16(textBox5.Text);

                Pen z = new Pen(bojaLinije, 10);
                g.DrawLine(z, x1,y1,x2,y2);
            }
            pictureBox1.Refresh();
        }

        private void pictureBox1_MouseDown(object sender, MouseEventArgs e)
        {
            if (e.Button == MouseButtons.Left)
            {
                crtam = true;
                prethodnatacka = e.Location;
                pocetnatacka = e.Location;
            }
        }

        private void pictureBox1_MouseMove(object sender, MouseEventArgs e)
        {
            if (crtam == false) return;
            if (comboBox1.Text == "Linija")
            {
                using (Graphics g = Graphics.FromImage(pictureBox1.BackgroundImage))
                {
                    using (Pen olovka = new Pen(bojaTacke, 5))
                    {
                        g.DrawLine(olovka, prethodnatacka, e.Location);
                    }
                    prethodnatacka = e.Location;
                    pictureBox1.Refresh();
                }
            }
           
        }

        private void pictureBox1_MouseUp(object sender, MouseEventArgs e)
        {
            if (crtam == false) return;
            crtam = false;
            if(comboBox1.Text=="Krug" || comboBox1.Text == "Pravougaonik")
            {
                using (Graphics g = Graphics.FromImage(pictureBox1.BackgroundImage))
                {
                    using (Pen olovka = new Pen(bojaLinije, 5))
                    {
                        int x = Math.Min(pocetnatacka.X, e.X);
                        int y = Math.Min(pocetnatacka.Y, e.Y);
                        int sirina = Math.Abs(pocetnatacka.X - e.X);
                        int visina = Math.Abs(pocetnatacka.Y - e.Y);
                        if (comboBox1.Text == "Krug")
                        {
                            g.DrawEllipse(olovka, x,y,sirina,visina);
                        }
                        if (comboBox1.Text == "Pravougaonik")
                        {
                            g.DrawRectangle(olovka, x, y, sirina, visina);
                        }
                    }
                   
                    pictureBox1.Refresh();
                }
            }
        }

        private void button4_Click(object sender, EventArgs e)
        {
            using (Graphics g = Graphics.FromImage(pictureBox1.BackgroundImage))
            {
                int x = Convert.ToInt16(textBox2.Text);
                int y = Convert.ToInt16(textBox3.Text);
                int sirina = Convert.ToInt16(textBox4.Text);
                int visina = Convert.ToInt16(textBox5.Text);

                using (Pen olovka = new Pen(bojaLinije, 5))
                {
                    g.DrawEllipse(olovka, x, y, sirina, visina);
                }
            }

            pictureBox1.Refresh();
        }

        private void button5_Click(object sender, EventArgs e)
        {
            using (Graphics g = Graphics.FromImage(pictureBox1.BackgroundImage))
            {
                int x = Convert.ToInt16(textBox2.Text);
                int y = Convert.ToInt16(textBox3.Text);
                int sirina = Convert.ToInt16(textBox4.Text);
                int visina = Convert.ToInt16(textBox5.Text);

                using (Pen olovka = new Pen(bojaLinije, 5))
                {
                    g.DrawRectangle(olovka, x, y, sirina, visina);
                }
            }

            pictureBox1.Refresh();
        }

        private void button6_Click(object sender, EventArgs e)
        {
            colorDialog1.Color = bojaLinije;

            if (colorDialog1.ShowDialog() == DialogResult.OK)
            {
                bojaLinije = colorDialog1.Color;
                button6.BackColor = bojaLinije;
            }
        }

        private void button7_Click(object sender, EventArgs e)
        {
            colorDialog1.Color = bojaTacke;

            if (colorDialog1.ShowDialog() == DialogResult.OK)
            {
                bojaTacke = colorDialog1.Color;
                button7.BackColor = bojaTacke;
            }
        }

        private void Form1_Load(object sender, EventArgs e)
        {
            if (comboBox1.Items.Count > 0)
            {
                comboBox1.SelectedIndex = 0;
            }
        }

        private void comboBox1_SelectedIndexChanged(object sender, EventArgs e)
        {

        }
    }
}
